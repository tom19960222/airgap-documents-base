## [Examples](aci_cloud_external_epg_selector_module.md#id5)

```yaml+jinja
- name: Add a new cloud external EPG selector
  cisco.aci.aci_cloud_external_epg_selector:
    host: apic
    username: admin
    password: SomeSecretPassword
    tenant: tenant1
    ap: ap1
    cloud_external_epg: ext_epg
    name: subnet_name
    subnet: 10.0.0.0/16
    state: present
  delegate_to: localhost

- name: Remove a cloud external EPG selector
  cisco.aci.aci_cloud_external_epg_selector:
    host: apic
    username: admin
    password: SomeSecretPassword
    validate_certs: false
    tenant: tenant1
    ap: ap1
    cloud_external_epg: ext_epg
    name: subnet_name
    subnet: 10.0.0.0/16
    state: absent
  delegate_to: localhost

- name: Query all cloud external EPG selectors
  cisco.aci.aci_cloud_external_epg_selector:
    host: apic
    username: admin
    password: SomeSecretPassword
    tenant: tenant1
    ap: ap1
    cloud_external_epg: ext_epg
    state: query
  delegate_to: localhost
```
